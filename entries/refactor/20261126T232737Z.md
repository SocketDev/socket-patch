#!/usr/bin/env python3
"""mkrepo.py <root> [catalog.json]: an npm monorepo fixture for the socket.yml policy."""
import hashlib, json, os, shutil, sys, base64
root = sys.argv[1]
def gh(b): return hashlib.sha256(b"blob %d\0" % len(b) + b).hexdigest()
shutil.rmtree(root, ignore_errors=True)
os.makedirs(os.path.join(root, ".git"))
open(os.path.join(root, ".git", "HEAD"), "w").write("ref: refs/heads/main\n")
# project dir -> [(name, version)]
PROJ = {
    "": [("lodash", "4.17.20"), ("left-pad", "1.3.0"), ("minimist", "1.2.5"), ("@babel/core", "7.0.0")],
    "services/payments": [("qs", "6.5.2")],
    "legacy": [("qs", "6.5.2"), ("ms", "2.0.0")],
    "tests/proj": [("ms", "2.0.0")],
}
# (name, version) -> [(uuid, severity, ghsa, tier, published)]
PATCHES = {
    ("lodash", "4.17.20"): [("aaaaaaaa-0000-4000-8000-000000000001", "critical", "GHSA-aaaa-aaaa-0001", "free", "2026-01-01T00:00:00Z")],
    ("left-pad", "1.3.0"): [("aaaaaaaa-0000-4000-8000-000000000002", "low", "GHSA-aaaa-aaaa-0002", "free", "2026-01-01T00:00:00Z")],
    ("minimist", "1.2.5"): [("aaaaaaaa-0000-4000-8000-000000000003", "high", "GHSA-aaaa-aaaa-0003", "free", "2026-01-01T00:00:00Z"),
                            ("aaaaaaaa-0000-4000-8000-000000000004", "medium", "GHSA-aaaa-aaaa-0004", "free", "2026-02-01T00:00:00Z")],
    ("@babel/core", "7.0.0"): [("aaaaaaaa-0000-4000-8000-000000000005", "moderate", "GHSA-aaaa-aaaa-0005", "free", "2026-01-01T00:00:00Z")],
    ("qs", "6.5.2"): [("aaaaaaaa-0000-4000-8000-000000000006", "high", "GHSA-aaaa-aaaa-0006", "free", "2026-01-01T00:00:00Z")],
    ("ms", "2.0.0"): [("aaaaaaaa-0000-4000-8000-000000000007", "", "GHSA-aaaa-aaaa-0007", "free", "2026-01-01T00:00:00Z")],
}
catalog = {}
for proj, pkgs in PROJ.items():
    d = os.path.join(root, proj)
    os.makedirs(d, exist_ok=True)
    deps = {n: v for n, v in pkgs}
    open(os.path.join(d, "package.json"), "w").write(json.dumps({"name": "p-" + (proj.replace("/", "-") or "root"), "version": "1.0.0", "dependencies": deps}, indent=2) + "\n")
    lock = {"name": "p", "version": "1.0.0", "lockfileVersion": 3, "requires": True,
            "packages": {"": {"name": "p", "version": "1.0.0", "dependencies": deps}}}
    for n, v in pkgs:
        base = n.split("/")[-1]
        lock["packages"]["node_modules/" + n] = {"version": v, "resolved": f"https://registry.npmjs.org/{n}/-/{base}-{v}.tgz", "integrity": "sha512-" + base64.b64encode(hashlib.sha512((n+v).encode()).digest()).decode()}
        pk = os.path.join(d, "node_modules", n)
        os.makedirs(pk)
        open(os.path.join(pk, "package.json"), "w").write(json.dumps({"name": n, "version": v}) + "\n")
        before = f"module.exports = '{n}@{v} before'\n".encode()
        open(os.path.join(pk, "index.js"), "wb").write(before)
        purl = f"pkg:npm/{n}@{v}"
        if purl not in catalog:
            ps = []
            for (uuid, sev, ghsa, tier, pub) in PATCHES[(n, v)]:
                after = f"module.exports = '{n}@{v} after {uuid[-1]}'\n".encode()
                ps.append({"uuid": uuid, "purl": purl, "tier": tier, "severity": sev, "ghsa": ghsa, "cve": "CVE-2026-" + uuid[-4:], "publishedAt": pub,
                           "files": {"package/index.js": {"before": base64.b64encode(before).decode(), "after": base64.b64encode(after).decode(),
                                     "beforeHash": gh(before), "afterHash": gh(after)}}})
            catalog[purl] = ps
    open(os.path.join(d, "package-lock.json"), "w").write(json.dumps(lock, indent=2) + "\n")
if len(sys.argv) > 2:
    json.dump(catalog, open(sys.argv[2], "w"), indent=1)

#!/usr/bin/env python3
"""api.py <catalog.json> <port> [log]: stand-in public patch proxy."""
import json, sys, base64, urllib.parse
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
cat = json.load(open(sys.argv[1])); port = int(sys.argv[2]); logf = open(sys.argv[3], "a") if len(sys.argv) > 3 else None
by_uuid = {p["uuid"]: p for ps in cat.values() for p in ps}
blobs = {}
for p in by_uuid.values():
    for f in p["files"].values():
        blobs[f["beforeHash"]] = base64.b64decode(f["before"]); blobs[f["afterHash"]] = base64.b64decode(f["after"])
def vulns(p):
    return {p["ghsa"]: {"cves": [p["cve"]], "summary": "s " + p["ghsa"], "severity": p["severity"], "description": "d"}}
def search(p):
    return {"uuid": p["uuid"], "purl": p["purl"], "publishedAt": p["publishedAt"], "description": "desc", "license": "MIT", "tier": p["tier"], "vulnerabilities": vulns(p)}
def base(purl):
    return purl.split("?")[0]
class H(BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"
    def log_message(self, *a): pass
    def send(self, code, obj, raw=None):
        body = raw if raw is not None else json.dumps(obj).encode()
        self.send_response(code); self.send_header("Content-Type", "application/octet-stream" if raw is not None else "application/json")
        self.send_header("Content-Length", str(len(body))); self.end_headers(); self.wfile.write(body)
    def note(self, extra=""):
        if logf: logf.write(f"{self.command} {self.path} AUTH={self.headers.get('Authorization')} {extra[:200]}\n"); logf.flush()
    def do_POST(self):
        n = int(self.headers.get("Content-Length", 0)); body = self.rfile.read(n)
        self.note(body.decode(errors="replace")[:2000])
        path = urllib.parse.urlparse(self.path).path
        if path.endswith("/batch"):
            purls = [c["purl"] for c in json.loads(body).get("components", [])]
            pk = []
            for u in purls:
                ps = cat.get(base(u))
                if ps:
                    pk.append({"purl": u, "patches": [{"uuid": p["uuid"], "purl": p["purl"], "tier": p["tier"], "cveIds": [p["cve"]], "ghsaIds": [p["ghsa"]], "severity": p["severity"] or None, "title": "t"} for p in ps]})
            return self.send(200, {"packages": pk, "canAccessPaidPatches": True})
        if path.endswith("/telemetry"):
            return self.send(200, {})
        return self.send(404, {"error": {"message": "no route"}})
    def do_GET(self):
        self.note()
        path = urllib.parse.urlparse(self.path).path
        parts = path.split("/")
        if path == "/v0/organizations":
            return self.send(200, {"organizations": {"1": {"id": "1", "name": "Auto", "image": None, "plan": "enterprise", "slug": "auto-org"}}})
        if "by-package" in parts:
            purl = urllib.parse.unquote(path.split("/by-package/", 1)[1])
            ps = cat.get(base(purl), [])
            return self.send(200, {"patches": [search(p) for p in ps], "canAccessPaidPatches": True})
        if "view" in parts:
            p = by_uuid.get(parts[-1])
            if not p: return self.send(404, {"error": {"message": "not found"}})
            r = search(p)
            r["files"] = {k: {"beforeHash": f["beforeHash"], "afterHash": f["afterHash"], "socketBlob": None, "blobContent": f["after"], "beforeBlobContent": f["before"]} for k, f in p["files"].items()}
            return self.send(200, r)
        if "blob" in parts:
            b = blobs.get(parts[-1])
            if b is None: return self.send(404, {"error": {"message": "not found"}})
            return self.send(200, None, raw=b)
        return self.send(404, {"error": {"message": "no route"}})
ThreadingHTTPServer(("127.0.0.1", port), H).serve_forever()
