### Cells
- **fail #372**: vlt ≥ 1.3.0 brotli tarballs. With a registry that advertises `dist.alternates` `tar.br`, vlt writes node flag bit 4. `vlt_lock_text.rs:673` only accepts 0–3, so hosted refuses the lock (`redirect_vlt_lock_unsupported`, rc 0, stays unpatched) and vendored fails with `vendor_lockfile_version_unsupported`. Reproduced on 1.3.1 and 1.3.2. The control with no alternates passes. I verified by hand that a fix must also clear bit 4, otherwise `vlt ci` rewrites the lock.
- pass, Linux, vlt 1.3.2: all five real-vlt capstones (`e2e_redirect_vlt_build` 37 ran / 1 skip, `e2e_vendor_vlt_build` 30/2, `mode_migration_vlt` 12/2, `e2e_safety_vlt` 7/3, `e2e_vlt` 14/1), and every `check-vlt-legs` manifest is clean. 1.3.2 behaves as 1.2.0 on npmjs bytes.

import http from 'node:http';
import zlib from 'node:zlib';
import crypto from 'node:crypto';
import { execSync } from 'node:child_process';
import fs from 'node:fs';
const PORT = +process.env.PORT || 18555;
const R = `http://127.0.0.1:${PORT}`;
const NAME='left-pad', VER='1.3.0', UUID='aaaaaaaa-aaaa-4aaa-8aaa-aaaaaaaaaaaa', TOKEN='bbbbbbbb-bbbb-4bbb-8bbb-bbbbbbbbbbbb', ORG='test-org';
const PURL=`pkg:npm/${NAME}@${VER}`;
const ALT = process.env.NO_ALT ? false : true;
function tarOf(content){
  const d=fs.mkdtempSync('/tmp/bt/t'); fs.mkdirSync(d+'/package');
  fs.writeFileSync(d+'/package/package.json', JSON.stringify({name:NAME,version:VER,main:'index.js'})+'\n');
  fs.writeFileSync(d+'/package/index.js', content);
  const tar = execSync(`tar -C ${d} --owner=0 --group=0 --mtime=@0 -cf - package`);
  return tar;
}
const sri = b => 'sha512-'+crypto.createHash('sha512').update(b).digest('base64');
const git = b => crypto.createHash('sha256').update(Buffer.concat([Buffer.from(`blob ${b.length}\0`), b])).digest('hex');
const PRIST = Buffer.from("module.exports = 'pristine'\n"), PATCH = Buffer.from("module.exports = 'patched'\n");
const tgz = zlib.gzipSync(tarOf(PRIST)), br = zlib.brotliCompressSync(tarOf(PRIST)), ptgz = zlib.gzipSync(tarOf(PATCH));
function wrapTar(){ const d=fs.mkdtempSync('/tmp/bt/w'); fs.mkdirSync(d+'/package'); fs.writeFileSync(d+'/package/package.json', JSON.stringify({name:'wrapper',version:'1.0.0',main:'index.js',dependencies:{'left-pad':'1.3.0'}})+'\n'); fs.writeFileSync(d+'/package/index.js', "module.exports = require('left-pad')\n"); return zlib.gzipSync(execSync(`tar -C ${d} --owner=0 --group=0 --mtime=@0 -cf - package`)); }
const wtgz = wrapTar();
const artPath = `/patch/npm/${NAME}/${VER}/${TOKEN}/${UUID}/${NAME}-${VER}.tgz`;
const log = [];
http.createServer((req,res)=>{
  let body=''; req.on('data',c=>body+=c); req.on('end',()=>{
  const u = req.url.split('?')[0]; log.push(req.method+' '+u); fs.appendFileSync('/tmp/bt/mock.log', req.method+' '+u+'\n');
  const j = o => { res.writeHead(200,{'content-type':'application/json'}); res.end(JSON.stringify(o)); };
  const b = x => { res.writeHead(200,{'content-type':'application/octet-stream'}); res.end(x); };
  if (u===`/${NAME}`) {
    const dist = {tarball:`${R}/${NAME}/-/${NAME}-${VER}.tgz`, integrity: sri(tgz)};
    if (ALT) dist.alternates=[{kind:'tar.br', tarball:`${R}/${NAME}/-/${NAME}-${VER}.tar.br`, integrity: sri(br)}];
    return j({name:NAME, 'dist-tags':{latest:VER}, versions:{[VER]:{name:NAME,version:VER,main:'index.js',dist}}});
  }
  if (u==='/wrapper') return j({name:'wrapper','dist-tags':{latest:'1.0.0'},versions:{'1.0.0':{name:'wrapper',version:'1.0.0',main:'index.js',dependencies:{'left-pad':'1.3.0'},dist:{tarball:`${R}/wrapper/-/wrapper-1.0.0.tgz`,integrity:sri(wtgz)}}}});
  if (u==='/wrapper/-/wrapper-1.0.0.tgz') return b(wtgz);
  if (u===`/${NAME}/-/${NAME}-${VER}.tgz`) return b(tgz);
  if (u===`/${NAME}/-/${NAME}-${VER}.tar.br`) return b(br);
  if (u===artPath) return b(ptgz);
  if (u.startsWith(`/v0/orgs/${ORG}/patches/blob/`)) { const h=u.split('/').pop(); for (const x of [PRIST,PATCH]) if (git(x)===h) return b(x); }
  if (u===`/v0/orgs/${ORG}/patches/batch`) return j({packages:[{purl:PURL,patches:[{uuid:UUID,purl:PURL,tier:'free',cveIds:[],ghsaIds:[],severity:'high',title:'fx'}]}],canAccessPaidPatches:false});
  if (u.startsWith(`/v0/orgs/${ORG}/patches/by-package/`)) return j({patches:[{uuid:UUID,purl:PURL,publishedAt:'2024-01-01T00:00:00Z',description:'x',license:'MIT',tier:'free',vulnerabilities:{}}],canAccessPaidPatches:false});
  if (u===`/v0/orgs/${ORG}/patches/package`) return j({results:{[UUID]:{status:'granted',url:R+artPath,purl:PURL,artifacts:[{kind:'tarball',url:R+artPath,integrity:{sha512:sri(ptgz)}}],registryOverride:null}}});

  res.writeHead(404); res.end('nf');
  });
}).listen(PORT, '127.0.0.1', ()=>console.log('listening', PORT));
